    avg_time = (total_run_duration / results['total_run']) if results['total_run'] > 0 else 0
    print(f"Average Time per Test: {avg_time:.2f} seconds")
    print(f"✅ Passed: {results['passed']}")
    print(f"🔴 Failed: {results['failed']}")
    print(f"🤔 Maybe (Info): {results['maybe']}")
    print("="*30)
    logger.info("--- Test Summary ---")
    logger.info(f"Total Run: {results['total_run']}, Passed: {results['passed']}, Failed: {results['failed']}, Maybe: {results['maybe']}, Skipped: {results['skipped']}")
    logger.info(f"Total Duration: {total_run_duration:.2f}s, Average Time: {avg_time:.2f}s")


    # --- Write results to CSV ---
    print("\n--- Writing Test Results to CSV ---")
    timestamp_str = datetime.datetime.now(datetime.timezone.utc).strftime("%Y%m%d_%H%M%S")
    results_filename = f"duration_test_run_{timestamp_str}.csv"
    results_filename_latest = "duration_test_run_latest.csv" # Overwrite latest

    if not all_results_data:
        print("No results data to write.")
        logger.warning("No results data generated to write to CSV.")
    else:
        # Dynamically determine fieldnames but prefer a specific order
        fieldnames_set = set()
        for row in all_results_data:
            fieldnames_set.update(row.keys())
        preferred_order = [
            "RunTimestamp", "TestID", "Category", "Description", "InputPrompt",
            "ExpectedOutcome", "ActualOutcome", "ResultStatus", "DurationSec",
            "ActualYears", "ActualMonths", "ActualTotalDurationYears", "ActualScore",
            "LLMRawOutputOverride", "Notes"
        ]
        # Ensure all keys are included, with preferred ones first
        fieldnames = [f for f in preferred_order if f in fieldnames_set] + \
                     sorted([f for f in fieldnames_set if f not in preferred_order])

        # Function to write CSV safely
        def write_csv(filename, data, field_names):
            try:
                with open(filename, 'w', newline='', encoding='utf-8') as csvfile:
                    writer = csv.DictWriter(csvfile, fieldnames=field_names, extrasaction='ignore')
                    writer.writeheader()
                    writer.writerows(data)
                print(f"Results successfully written to {filename}")
                logger.info(f"Test results successfully written to {filename}")
            except IOError as e:
                print(f"Error writing results to CSV ({filename}): {e}")
                logger.error(f"Failed to write results CSV ({filename}): {e}")
            except Exception as e:
                 print(f"An unexpected error occurred writing CSV ({filename}): {e}")
                 logger.exception(f"Unexpected error writing results CSV ({filename})")

        # Write timestamped file
        write_csv(results_filename, all_results_data, fieldnames)
        # Write/overwrite latest file
        write_csv(results_filename_latest, all_results_data, fieldnames)

    print("--- Testing Complete ---")
    logger.info("--- Duration Bot Testing Complete ---")
