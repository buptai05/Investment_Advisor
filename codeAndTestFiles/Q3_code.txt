# --- Configuration Class ---
@dataclass
class Config:
    """Configuration for the income percentage scoring system"""
    model_id: str = "mistralai/Mistral-7B-Instruct-v0.3" # Or your preferred model
    max_attempts: int = 3
    min_valid_percentage: float = 0.0
    max_valid_percentage: float = 100.0
    percentage_score_brackets: List[Tuple[float, float, int]] = field(default_factory=lambda: [
        (50.0, float('inf'), 50), # Percentage > 50 (Note: upper bound is exclusive)
        (25.0, 50.0 + 1e-9, 40), # 25 < Percentage <= 50
        (10.0, 25.0 + 1e-9, 30), # 10 < Percentage <= 25
        (0.0 + 1e-9, 10.0 + 1e-9, 20), # 0 < Percentage <= 10
        (0.0, 0.0 + 1e-9, 10),   # Percentage == 0 (handle exact zero)
        # Note: Using small epsilon (1e-9) to handle boundary conditions with floats.
        # The logic becomes: score if lower_exclusive < value <= upper_inclusive
        # For >= 50, we use (50.0, inf, 50) meaning 50 < value < inf gets score 50
        # To make it >= 50, we should adjust or use a different structure.
        # Let's redefine brackets for clarity: [lower_inclusive, upper_exclusive) -> score
        # (50.0, float('inf'), 50), # Percentage >= 50
        # (25.0, 50.0, 40),        # 25 <= Percentage < 50
        # (10.0, 25.0, 30),        # 10 <= Percentage < 25
        # (0.0, 10.0, 20),         # 0 < Percentage < 10 (Need to handle > 0 separately if needed)
        # (0.0, 0.0 + 1e-9, 10),   # Special case for exactly 0, if desired.
        # Simpler logic based on original code:
        # Percentage > 50 -> 50
        # 25 < P <= 50 -> 40
        # 10 < P <= 25 -> 30
        # 0 < P <= 10 -> 20
        # P == 0 -> 10
    ])
    # Re-simplifying brackets based on original logic: lower bound (inclusive), upper bound (exclusive), score
    percentage_score_brackets_simple: List[Tuple[float, float, int]] = field(default_factory=lambda: [
        (50.0 + 1e-9, float('inf'), 50), # > 50%
        (25.0 + 1e-9, 50.0 + 1e-9, 40),  # > 25% to <= 50%
        (10.0 + 1e-9, 25.0 + 1e-9, 30),  # > 10% to <= 25%
        (0.0 + 1e-9, 10.0 + 1e-9, 20),   # > 0% to <= 10%
        (0.0, 0.0 + 1e-9, 10)            # Exactly 0%
    ])
    llm_params: Dict[str, Any] = field(default_factory=lambda: {
        "temperature": 0.1,
        "max_new_tokens": 10, # Shorter response needed
        "top_k": 5,
        "repetition_penalty": 1.1,
    })

# --- Logging Setup ---
def setup_logging(level=logging.INFO) -> logging.Logger:
    """Configure logging for the application"""
    logging.basicConfig(
        level=level,
        format='%(asctime)s - %(name)s - %(levelname)s - %(message)s',
        datefmt='%Y-%m-%d %H:%M:%S'
    )

    logging.getLogger("httpcore").setLevel(logging.ERROR)
    logging.getLogger("huggingface_hub").setLevel(logging.ERROR)
    logging.getLogger("transformers").setLevel(logging.ERROR)
    return logging.getLogger("IncomePercentageBot")

logger = setup_logging()

# --- Validation ---
def validate_parsed_percentage(percentage: float, config: Config) -> bool:
    """Validate that parsed percentage is within the configured range."""
    if not (config.min_valid_percentage <= percentage <= config.max_valid_percentage):
        logger.warning(
            f"Parsed percentage {percentage:.2f}% is outside the valid range "
            f"({config.min_valid_percentage}%-{config.max_valid_percentage}%)."
        )
        return False
    return True

# --- Scoring ---
def calculate_percentage_score(percentage: float, brackets: List[Tuple[float, float, int]]) -> Optional[int]:
    """Determine score based on percentage brackets. Returns None if no bracket matches."""
    # Using the simplified brackets definition matching original logic
    for lower_inclusive, upper_exclusive, score in brackets:
        # Check if percentage falls within the bracket [lower, upper)
        if lower_inclusive <= percentage < upper_exclusive:
            logger.debug(f"Percentage {percentage:.2f}% matched bracket [{lower_inclusive}, {upper_exclusive}) -> Score: {score}")
            return score
    logger.error(f"Percentage {percentage:.2f}% did not fall into any defined score bracket. Check bracket definitions.")
    return None # Explicitly return None if no match

# get_huggingface_token remains the same as in the age example
def get_huggingface_token() -> Optional[str]:
    """Retrieves the Hugging Face API token from environment, Colab, or prompt."""
    hf_token = os.environ.get("HUGGINGFACE_API_TOKEN")
    if hf_token:
        logger.info("Hugging Face token found in environment variables.")
        return hf_token
    try:
        # Try Colab userdata first if in Colab environment
        from google.colab import userdata
        hf_token = userdata.get('HUGGINGFACE_API_TOKEN')
        if hf_token:
            logger.info("Hugging Face token found in Colab userdata.")
            return hf_token
    except (ImportError, ModuleNotFoundError):
        logger.debug("google.colab module not found, skipping userdata check.")
    except KeyError:
        logger.info("Hugging Face token not found in Colab userdata.")

    # Fallback to getpass if not found elsewhere
    logger.warning("Hugging Face token not found in environment or Colab userdata.")
    print("Please enter your Hugging Face API token (needed for the LLM):")
    try:
        hf_token = getpass("    > ")
        if hf_token:
            logger.info("Hugging Face token received via prompt.")
            print("Tip: Set the HUGGINGFACE_API_TOKEN environment variable for automatic use next time.")
            return hf_token
        else:
            logger.error("No Hugging Face token provided via prompt.")
            return None
    except EOFError:
         logger.error("Could not read Hugging Face token from input (EOF encountered).")
         return None


def setup_llm_chain(config: Config):
    """Sets up the LangChain LLM chain for income percentage extraction."""
    from langchain_huggingface import HuggingFaceEndpoint
    from langchain.chains import LLMChain
    from langchain_core.prompts import PromptTemplate

    hf_token = get_huggingface_token()
    if not hf_token:
        raise ValueError("Hugging Face API token is required but could not be obtained.")

    try:
        llm = HuggingFaceEndpoint(
            repo_id=config.model_id,
            task="text-generation",
            huggingfacehub_api_token=hf_token,
            # Add stop_sequences here:
            stop_sequences=["\n", "User Reply:", "AI Output:"], # Stop on newline or if it tries to mimic the prompt
            **config.llm_params
        )
    except Exception as e:
        logger.exception(f"Failed to initialize HuggingFaceEndpoint for model {config.model_id}: {e}")
        raise ConnectionError(f"Could not connect to Hugging Face Endpoint: {e}") from e

    # --- Enhanced Prompt Template for Percentage Extraction ---
    income_percentage_template = """You are an AI assistant specialized in extracting numerical data.
The user was asked: "What percentage of your annual household income could be available for investment or savings?"
Your task is to analyze the user's reply and extract *only* the numerical percentage value they stated.

RULES:
1. Focus *only* on numbers mentioned as percentages related to the question (often near words like '%', 'percent', 'save', 'invest'). Convert percentage words (like 'twenty') to numbers.
2. Ignore any '%' signs or the word 'percent' in the final output.
3. Output *only* the number itself (e.g., "15", "22.5", "0").
4. If the user's reply does not contain a clear numerical percentage relevant to the question (e.g., they say "I don't know", "a small amount", "maybe invest in stocks", use vague ranges like "10-15%"), output the exact word: None
5. Ignore other numbers in the text like salaries, ages, amounts in dollars, etc. Focus ONLY on the SAVINGS/INVESTMENT PERCENTAGE.

CRITICAL: Your response MUST be *either* the extracted number as a string *or* the literal word "None". Do NOT include any other text, explanation, or formatting.

Examples:
User Reply: "About 35%"
AI Output: 35
User Reply: "Roughly 12.5 percent"
AI Output: 12.5
User Reply: "I can save 50 %"
AI Output: 50
User Reply: "Zero for now"
AI Output: 0
User Reply: "Maybe 5 or 10"
AI Output: None
User Reply: "I don't know"
AI Output: None
User Reply: "Maybe stocks and mutual funds"
AI Output: None
User Reply: "My goal is 100 dollars"
AI Output: None
User Reply: "Around twenty percent I think"
AI Output: 20
User Reply: "My salary is 90000 per year, I think I can put aside 10% for savings."
AI Output: 10
User Reply: "my age is 10 but i can save twenty percent"
AI Output: 20

User Reply: {user_reply}
AI Output:"""


    prompt = PromptTemplate(template=income_percentage_template, input_variables=["user_reply"])

    try:
        percentage_chain = LLMChain(prompt=prompt, llm=llm, verbose=False)
        logger.info(f"LLM Chain initialized successfully for percentage extraction with model {config.model_id}.")
        return percentage_chain
    except Exception as e:
        logger.exception(f"Failed to create LLMChain: {e}")
        raise RuntimeError(f"Could not create LLMChain: {e}") from e


# %% [Main Application Class]
class IncomePercentageBot:
    """Manages the income percentage query, LLM interaction, parsing, and scoring."""

    def __init__(self, config: Optional[Config] = None):
        """Initializes the bot with configuration."""
        self.config = config or Config()
        self._llm_chain = None
        self._llm_initialized = False
        logger.info("IncomePercentageBot initialized.")
        logger.debug(f"Configuration: {self.config}")

    def _initialize_llm_chain(self):
        """Initializes the LLM chain if not already done."""
        if not self._llm_initialized:
            logger.info("Attempting to initialize LLM chain...")
            try:
                self._llm_chain = setup_llm_chain(self.config)
                self._llm_initialized = True
            except (ValueError, ConnectionError, RuntimeError) as e:
                logger.error(f"LLM Chain initialization failed: {e}")
                # Don't raise here, allow run_interaction to handle it gracefully
                # raise # Re-raise if initialization is critical before interaction starts

    @property
    def llm_chain(self):
        """Property to lazily initialize and return the LLM chain."""
        if not self._llm_initialized:
            self._initialize_llm_chain()
        # Check again after attempting initialization
        if not self._llm_chain:
            raise RuntimeError("LLM Chain could not be initialized. Please check token and connection.")
        return self._llm_chain

    def parse_llm_percentage_response(self, llm_text_output: str) -> Optional[float]:
        """
        Parses the LLM's text output expecting a number string or 'None'.
        Returns the float value of the percentage or None if 'None' is received or parsing fails.
        """
        llm_text_output = llm_text_output.strip()
        logger.debug(f"Attempting to parse LLM output: '{llm_text_output}'")

        if llm_text_output.lower() == 'none':
            logger.info("LLM indicated no valid percentage found ('None').")
            return None

        # Try to convert the remaining output to a float
        try:
            # Basic regex to find a number (int or float, potentially negative but validation handles range)
            # This helps strip potential minor LLM preamble/postamble if the prompt is slightly ignored
            match = re.search(r'^(-?\d+\.?\d*)$', llm_text_output)
            if match:
                percentage = float(match.group(1))
                logger.info(f"Successfully parsed percentage from LLM: {percentage}")
                return percentage
            else:
                # Maybe the LLM included text despite instructions
                # Try finding a number within the string more broadly
                fallback_match = re.search(r'(-?\d+\.?\d*)', llm_text_output)
                if fallback_match:
                     percentage = float(fallback_match.group(1))
                     logger.warning(f"LLM response '{llm_text_output}' wasn't clean, but extracted number: {percentage}")
                     return percentage
                else:
                    logger.error(f"Failed to parse numeric value from LLM response: '{llm_text_output}'. Not a valid number or 'None'.")
                    return None

        except ValueError:
            logger.error(f"Could not convert LLM response '{llm_text_output}' to a float.")
            return None
        except Exception as e:
            logger.exception(f"An unexpected error occurred during parsing LLM response '{llm_text_output}': {e}")
            return None


    def process_user_input(self, user_input: str) -> Optional[float]:
        """
        Processes user input using LLM, parses response, validates percentage.
        Returns the validated percentage (float) or None if processing fails.
        """
        if not user_input:
            logger.warning("Received empty user input.")
            return None

        try:
            # Ensure LLM is ready (will attempt init if needed)
            llm_chain_instance = self.llm_chain
            logger.info(f"Sending user input to LLM for percentage extraction: '{user_input}'")

            # Invoke the LLM chain
            llm_response = llm_chain_instance.invoke({"user_reply": user_input})
            llm_text_output = llm_response.get('text', '').strip() if isinstance(llm_response, dict) else str(llm_response).strip()

            if not llm_text_output:
                logger.error("Received empty response from LLM.")
                return None
            logger.info(f"Received LLM response: '{llm_text_output}'")

            # Parse the LLM response
            parsed_percentage = self.parse_llm_percentage_response(llm_text_output)

            if parsed_percentage is None:
                # Parsing failed or LLM returned 'None'
                # parse_llm_percentage_response already logged the reason
                return None # Indicate failure

            # Validate the parsed percentage
            if not validate_parsed_percentage(parsed_percentage, self.config):
                # Validation function already logged the warning
                return None # Indicate validation failure

            # If parsing and validation succeeded
            logger.info(f"Successfully processed input. Validated percentage: {parsed_percentage:.2f}%.")
            return parsed_percentage

        except RuntimeError as e:
             # Catch errors during lazy init if it fails
             logger.error(f"LLM Chain Runtime Error: {e}")
             raise # Re-raise to be caught by run_interaction's handler
        except Exception as e:
            # Catch unexpected errors (e.g., LLM API call failure during invoke)
            logger.exception(f"An unexpected error occurred during LLM interaction or processing: {e}")
            return None # Indicate failure


    def run_interaction(self) -> Optional[int]:
        """Runs the interactive command-line loop for the percentage bot."""
        print("\n" + "="*45) # Slightly wider line
        print(" Income Percentage for Investment/Savings")
        print("="*45)
        print("\nLet's determine a score based on potential savings.")

        # Attempt to initialize LLM early and handle failure gracefully
        try:
            self._initialize_llm_chain()
        except (ValueError, ConnectionError, RuntimeError) as init_error:
            print(f"\n[Error] Sorry, I couldn't set up the connection needed: {init_error}")
            print("Please check your setup (API token, network) and try again later.")
            return None # Cannot proceed without LLM

        attempts = 0
        final_score = None

        while attempts < self.config.max_attempts:
            current_attempt = attempts + 1
            try:
                # --- Clearer Prompt ---
                if attempts == 0:
                    prompt_message = "What percentage of your annual household income could be available for investment or savings? (e.g., 10%, 25.5, 0)"
                else:
                    prompt_message = f"Let's try again. Please state the percentage (0-{int(self.config.max_valid_percentage)}) you could save or invest:"

                print(f"\n[{current_attempt}/{self.config.max_attempts}] {prompt_message}")
                user_input = input("    > ").strip()

                if not user_input:
                    print("Please enter a value.")
                    # Don't count empty input as a full attempt unless desired
                    continue # Ask again without incrementing attempts

                # Process the input
                percentage_result = self.process_user_input(user_input)

                if percentage_result is not None:
                    # Successfully parsed and validated a percentage
                    print(f"\nUnderstood. The percentage is {percentage_result:.2f}%.")

                    # Calculate score using the simplified brackets from Config
                    score = calculate_percentage_score(percentage_result, self.config.percentage_score_brackets_simple)

                    if score is not None:
                        print(f"Based on this, the score is: {score}")
                        final_score = score
                        break # Success
                    else:
                        # This should theoretically not happen if brackets cover all valid ranges
                        print("[Internal Error] Couldn't calculate a score for that percentage.")
                        logger.error(f"Scoring error for validated percentage {percentage_result:.2f}. Check bracket logic.")
                        # Treat as a failed attempt if scoring fails unexpectedly
                        attempts += 1

                else:
                    # process_user_input returned None (LLM failed, parsing failed, or validation failed)
                    print("\nHmm, I couldn't understand that as a percentage between "
                          f"{int(self.config.min_valid_percentage)}% and {int(self.config.max_valid_percentage)}%.")
                    attempts += 1 # Increment attempt counter on failure

            except KeyboardInterrupt:
                print("\n\nOkay, cancelling this session.")
                return None
            except EOFError:
                print("\n\nInput ended unexpectedly.")
                return None
            except RuntimeError as e:
                 # Catch LLM init errors re-raised from process_user_input
                 logger.error(f"LLM Runtime error during interaction: {e}")
                 print(f"\n[Error] A problem occurred with the AI service: {e}")
                 print("Please check your setup and try again later.")
                 return None # Critical error, stop interaction
            except Exception as e:
                 logger.exception(f"An unexpected error occurred in the interaction loop: {e}")
                 print(f"\nAn unexpected issue occurred ({type(e).__name__}). Please try again.")
                 # Increment attempts for unexpected errors too, to prevent infinite loops
                 attempts += 1

        # After the loop
        if final_score is None and attempts >= self.config.max_attempts:
             print(f"\nSorry, I couldn't determine a valid percentage after {self.config.max_attempts} tries.")
             print("We'll have to skip this question.")

        return final_score # Return the score (or None)

# %% [Main Execution]
if __name__ == "__main__":
    bot_instance = None
    final_result = None
    try:
        # Setup config and bot instance
        app_config = Config()
        bot_instance = IncomePercentageBot(config=app_config)
        # Run the interaction loop
        final_result = bot_instance.run_interaction()

        # Print final outcome
        if final_result is not None:
            print(f"\n--- Interaction Complete. Final Score: {final_result} ---")
        else:
            print("\n--- Interaction ended without determining a final score. ---")

    except Exception as main_error:
         # Catch critical errors during setup or execution outside the loop
         logger.exception(f"A critical error occurred in the main execution block: {main_error}")
         print(f"\nA critical error occurred: {main_error}")

    finally:
        print("\nExiting Income Percentage Bot.")
