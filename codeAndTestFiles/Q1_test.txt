            run_this_test = False

        if not run_this_test:
            # print(f"\n--- Skipping Test Case {i+1}/{total_tests} [{test_id}] ---")
            continue
        # --- End Filtering ---


        print(f"\n--- Test Case {i+1}/{total_tests} [{test_id} / {test['category']}] ---")
        print(f"Description: {test['description']}")
        display_prompt = test['prompt'].format(current_year=current_year)
        print(f"Input: \"{display_prompt}\"")
        print(f"Expected Outcome Type: {test['expected']}")

        actual_score = None
        age_result_tuple = None
        duration = -1.0
        age_details = "N/A"
        outcome_type = 'fail'

        # --- Execute the core logic with Timing ---
        try:
            start_time = time.monotonic()
            age_result_tuple = chatbot_instance.process_user_input(test['prompt'])
            duration = time.monotonic() - start_time

            if age_result_tuple:
                years, months, total_age = age_result_tuple
                age_details = f"Processed: (Years={years:.2f}, Months={months:.2f}, Total={total_age:.2f})"
                score = calculate_age_score(total_age, chatbot_instance.config.age_brackets)
                if score is not None:
                    actual_score = score
                    outcome_type = 'score'
                    age_details += f" -> Score: {score}"
                else:
                    age_details += " | Scoring Failed (returned None)"
                    outcome_type = 'fail'
            else:
                age_details = "process_user_input returned None (Failure signal)"
                outcome_type = 'fail'

        except Exception as e:
            duration = time.monotonic() - start_time if 'start_time' in locals() else -1.0
            logger.error(f"Exception during test case {i+1} ('{test['description']}'): {e}")
            print(f"  ERROR: Exception occurred during processing: {e}")
            outcome_type = 'fail'
            age_details = f"Exception: {type(e).__name__}: {str(e)[:100]}..."
        # --- End Core Logic ---


        # --- Evaluate Result ---
        result_status = "FAIL"
        log_prefix = "🔴"

        if test['expected'] == outcome_type:
            result_status = "PASS"
            log_prefix = "✅"
            results['passed'] += 1
        elif test['expected'] == 'maybe_score':
            result_status = f"INFO (Maybe - {outcome_type.capitalize()})"
            log_prefix = "🤔"
            results['maybe'] += 1
        else:
            results['failed'] += 1

        print(f"  {log_prefix} Result: {result_status}")
        print(f"      Actual Outcome Type: {outcome_type}")
        print(f"      Duration: {duration:.4f} sec")
        print(f"      Age Details: {age_details}")
        if actual_score is not None:
             print(f"      Actual Score: {actual_score}")

        # --- Store result for CSV ---
        result_row = {
            "RunTimestamp": datetime.datetime.now(datetime.timezone.utc).isoformat(),
            "TestID": test_id,
            "Category": test['category'],
            "Description": test['description'],
            "InputPrompt": test['prompt'],
            "ExpectedOutcome": test['expected'],
            "ActualOutcome": outcome_type,
            "ResultStatus": result_status.split(" ")[0],
            "DurationSec": f"{duration:.4f}",
            "ActualAgeTuple": f"({age_result_tuple[0]:.2f}, {age_result_tuple[1]:.2f})" if age_result_tuple else "",
            "ActualScore": str(actual_score) if actual_score is not None else "",
            "Notes": age_details
        }
        all_results_data.append(result_row)

        # Optional: Add a small delay between API calls if needed
        # time.sleep(0.1)

    # --- Print Summary ---
    print("\n" + "="*30)
    print("--- Test Summary ---")
    print(f"Total Tests Run: {results['passed'] + results['failed'] + results['maybe']} / {total_tests}")
    print(f"✅ Passed: {results['passed']}")
    print(f"🔴 Failed: {results['failed']}")
    print(f"🤔 Maybe (Info): {results['maybe']}")
    print("="*30)

    # --- Write results to CSV ---
    print("\n--- Writing Test Results to CSV ---")
    results_filename = "test_run_latest.csv"
    if not all_results_data:
        print("No results data to write.")
    else:
        fieldnames_set = set()
        for row in all_results_data:
            fieldnames_set.update(row.keys())
        preferred_order = ["RunTimestamp", "TestID", "Category", "Description", "InputPrompt",
                           "ExpectedOutcome", "ActualOutcome", "ResultStatus", "DurationSec",
                           "ActualAgeTuple", "ActualScore", "Notes"]
        fieldnames = [f for f in preferred_order if f in fieldnames_set] + \
                     sorted([f for f in fieldnames_set if f not in preferred_order])

        try:
            with open(results_filename, 'w', newline='', encoding='utf-8') as csvfile:
                writer = csv.DictWriter(csvfile, fieldnames=fieldnames, extrasaction='ignore')
                writer.writeheader()
                writer.writerows(all_results_data)
            print(f"Results successfully written to {results_filename}")
        except Exception as e:
            print(f"Error writing results to CSV: {e}")
            logger.error(f"Failed to write results CSV: {e}")

    print("--- Testing Complete ---")
