        except Exception as e:
            duration = time.monotonic() - start_time if 'start_time' in locals() else -1.0
            logger.error(f"Exception during test case {test_id} ('{test['description']}'): {e}", exc_info=True)

            actual_outcome_category = 'fail' # Exception means failure

                    log_prefix = "✅"

            else:

                log_prefix = "✅"

        else:
            # Categories did not match (and not a 'maybe' case)
            result_status = f"FAIL (Category Mismatch - Got {actual_outcome_category}, Expected {expected_cat})"
            results_summary['failed'] += 1


        print(f"  {log_prefix} Result: {result_status}")
        print(f"      Actual Outcome Category: {actual_outcome_category}")
        print(f"      Duration: {duration:.4f} sec")
        print(f"      Processing Details: {processing_details}")
        if actual_score is not None:

        fieldnames = [f for f in preferred_order if f in fieldnames_set] + \
                     sorted([f for f in fieldnames_set if f not in preferred_order])

        except Exception as e:
            print(f"Error writing results to CSV: {e}")
            logger.error(f"Failed to write results CSV: {e}")

    print("--- Stage of Life Testing Complete ---")
