        actual_score: Optional[int] = None
        duration: float = -1.0
        processing_details: str = "N/A"
        actual_outcome_category: ExpectedOutcomeCategory = 'fail' # Default to fail

            start_time = time.monotonic()
            # Use the financial familiarity chatbot instance's processing method
            actual_score = chatbot_instance.process_user_input(test['prompt'])
            duration = time.monotonic() - start_time

            # Determine actual outcome category based on result

                     # This case might occur if parse_llm response has issues despite checks

                     processing_details = f"ERROR: process_user_input returned invalid score: {actual_score} (Valid: {valid_scores})"
                     logger.error(f"Test {test_id}: process_user_input returned invalid score {actual_score}, valid are {valid_scores}")

            else:
                # process_user_input returns None if LLM returned 'none', parsing failed, or input was empty/whitespace
                # Check if the input was valid before assuming 'none' from LLM
                if not test['prompt'] or test['prompt'].isspace():
                    actual_outcome_category = 'fail' # Empty/whitespace input is a test harness failure
                    processing_details = "Input was empty or whitespace"
                else:
                    actual_outcome_category = 'none' # Assume 'none' from LLM/parser otherwise
                    processing_details = "process_user_input returned None (Indicates 'none', parse failure, or invalid input)"


        except Exception as e:
            duration = time.monotonic() - start_time if 'start_time' in locals() else -1.0
            logger.error(f"Exception during test case {test_id} ('{test['description']}'): {e}", exc_info=True)
            print(f"  ERROR: Exception occurred during processing: {e}")
            actual_outcome_category = 'fail' # Exception means failure
            processing_details = f"Exception: {type(e).__name__}: {str(e)[:100]}..."
            actual_score = None # Ensure score is None on exception
        # --- End Core Logic ---

        # --- Evaluate Result ---
        result_status = "FAIL"
        log_prefix = "🔴"

        expected_cat = test['expected_outcome_category']
        expected_score_val = test['expected_score']

        if expected_cat == actual_outcome_category:
            # Categories match, now check score if expected category is 'score'
            if expected_cat == 'score':
                if actual_score == expected_score_val:
                    result_status = "PASS"
                    log_prefix = "✅"
                    results_summary['passed'] += 1
                else:
                    # Category matched ('score') but the score value itself was wrong
                    result_status = f"FAIL (Wrong Score - Got {actual_score}, Expected {expected_score_val})"
                    results_summary['failed'] += 1
            elif expected_cat == 'fail':
                 # Category matched ('fail') - This is expected for certain inputs (e.g., empty)
                 result_status = "PASS"
                 log_prefix = "✅"
                 results_summary['passed'] += 1
            else: # 'none'
                # Categories matched ('none')
                result_status = "PASS"
                log_prefix = "✅"
                results_summary['passed'] += 1
        elif expected_cat == 'maybe_score':
             # For 'maybe_score', accept 'score' (with any valid score) or 'none' as plausible outcomes
             if actual_outcome_category == 'score' or actual_outcome_category == 'none':
                 result_status = f"INFO (Maybe - Plausible Outcome: {actual_outcome_category.capitalize()})"
                 if actual_outcome_category == 'score':
                      result_status += f" - Score: {actual_score}" # Add score if available
                 log_prefix = "🤔"
                 results_summary['maybe'] += 1
             else: # If it failed unexpectedly
                 result_status = "FAIL (Expected Maybe, Got Fail)"
                 results_summary['failed'] += 1
        else:
            # Categories did not match (and not a 'maybe' case)
            result_status = f"FAIL (Category Mismatch - Got {actual_outcome_category}, Expected {expected_cat})"
            results_summary['failed'] += 1


        print(f"  {log_prefix} Result: {result_status}")
        print(f"      Actual Outcome Category: {actual_outcome_category}")
        print(f"      Duration: {duration:.4f} sec")
        print(f"      Processing Details: {processing_details}")
        if actual_score is not None:
             print(f"      Actual Score: {actual_score}")


        # --- Store result for CSV ---
        result_row: Dict[str, Any] = {
            "RunTimestamp": datetime.datetime.now(datetime.timezone.utc).isoformat(),
            "TestID": test_id,
            "Category": test['category'],
            "Description": test['description'],
            "InputPrompt": test['prompt'],
            "ExpectedOutcomeCategory": test['expected_outcome_category'],
            "ExpectedScore": str(test['expected_score']) if test['expected_score'] is not None else "",
            "ActualOutcomeCategory": actual_outcome_category,
            "ActualScore": str(actual_score) if actual_score is not None else "",
            "ResultStatus": result_status.split(" ")[0], # Get main status (PASS/FAIL/INFO)
            "DurationSec": f"{duration:.4f}",
            "Notes": processing_details if result_status.startswith("FAIL") or result_status.startswith("INFO") else "" # Add details on fail/info
        }
        all_results_data.append(result_row)

        # Optional: Add a small delay between API calls if needed
        # time.sleep(0.1)

    # --- Print Summary ---
    print("\n" + "="*40)
    print("--- Financial Familiarity Test Summary ---")

    # Use a specific filename for financial familiarity results
    results_filename = "financial_familiarity_test_run_latest.csv"
    if not all_results_data:
        print("No results data to write.")

            "ResultStatus", "DurationSec", "Notes"
        ]

        fieldnames_set = set()
        for row in all_results_data:
            fieldnames_set.update(row.keys())
        fieldnames = [f for f in preferred_order if f in fieldnames_set] + \
                     sorted([f for f in fieldnames_set if f not in preferred_order])

        try:
            with open(results_filename, 'w', newline='', encoding='utf-8') as csvfile:
                writer = csv.DictWriter(csvfile, fieldnames=fieldnames, extrasaction='ignore')
                writer.writeheader()
                writer.writerows(all_results_data)
            print(f"Results successfully written to {results_filename}")
        except Exception as e:
            print(f"Error writing results to CSV: {e}")
            logger.error(f"Failed to write results CSV: {e}")

    print("--- Financial Familiarity Testing Complete ---")
