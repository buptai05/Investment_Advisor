                 result_status = f"INFO (Maybe - Plausible Outcome: {actual_outcome_category.capitalize()})"
                 if actual_outcome_category == 'score':
                      result_status += f" Score: {actual_score}" # Add the score obtained in maybe cases
                 log_prefix = "🤔"
                 results_summary['maybe'] += 1
             else: # If it failed unexpectedly (e.g., exception or invalid score)
                 result_status = f"FAIL (Expected Maybe, Got {actual_outcome_category.capitalize()})"
                 results_summary['failed'] += 1
        else: # Categories mismatch (and not maybe)
            # Example: Expected 'score', got 'none' OR Expected 'none', got 'score' OR Expected 'score', got 'fail'
            result_status = f"FAIL (Category Mismatch - Got {actual_outcome_category}, Expected {expected_cat})"
            if actual_outcome_category == 'score' and expected_cat != 'score':
                 result_status += f" Unexpected Score: {actual_score}" # Add score info if mismatch involves getting a score unexpectedly
            results_summary['failed'] += 1


        print(f"  {log_prefix} Result: {result_status}")
        print(f"      Actual Outcome Category: {actual_outcome_category}")
        print(f"      Duration: {duration:.4f} sec")
        print(f"      Processing Details: {processing_details}")
        if actual_score is not None and actual_outcome_category == 'score': # Only print score if category is score
             print(f"      Actual Score: {actual_score}")


        # --- Store result for CSV ---
        result_row: Dict[str, Any] = {
            "RunTimestamp": datetime.datetime.now(datetime.timezone.utc).isoformat(),
            "TestID": test_id,
            "Category": test['category'],
            "Description": test['description'],
            "InputPrompt": test['prompt'],
            "ExpectedOutcomeCategory": test['expected_outcome_category'],
            "ExpectedScore": str(test['expected_score']) if test['expected_score'] is not None else "",
            "ActualOutcomeCategory": actual_outcome_category,
            "ActualScore": str(actual_score) if actual_score is not None else "",
            "ResultStatus": result_status.split(" ")[0], # Get main status (PASS/FAIL/INFO)
            "DurationSec": f"{duration:.4f}",
            "Notes": processing_details if result_status.startswith("FAIL") or result_status.startswith("INFO") else result_status # Add details on fail/info, include score if needed
        }
        # Refine notes for maybe cases to include score if available
        if result_status.startswith("INFO (Maybe") and actual_score is not None:
            result_row["Notes"] = f"Plausible Outcome: Score: {actual_score}"
        elif result_status.startswith("INFO (Maybe") and actual_score is None:
             result_row["Notes"] = f"Plausible Outcome: None"


        all_results_data.append(result_row)

        # Optional: Add a small delay between API calls if needed to avoid rate limits
        # time.sleep(0.1)

    # --- Print Summary ---
    print("\n" + "="*30)
    print("--- Test Summary ---")
    total_run = results_summary['passed'] + results_summary['failed'] + results_summary['maybe']
    # Filter out skipped tests if filtering was applied
    skipped_tests = total_tests - total_run if (categories_to_run or ids_to_run) else 0
    print(f"Total Tests Defined: {total_tests}")
    if skipped_tests > 0:
         print(f"Total Tests Skipped: {skipped_tests}")
    print(f"Total Tests Run: {total_run}")
    print(f"✅ Passed: {results_summary['passed']}")
    print(f"🔴 Failed: {results_summary['failed']}")
    print(f"🤔 Maybe (Info): {results_summary['maybe']}")
    print("="*30)

    # --- Write results to CSV ---
    print("\n--- Writing Test Results to CSV ---")
    # Use a specific filename for financial familiarity results
    results_filename = "financial_familiarity_test_run_latest.csv"
    if not all_results_data:
        print("No results data to write (perhaps all tests were skipped or an error occurred before running).")
    else:
        # Define preferred order for CSV columns
        preferred_order = [
            "RunTimestamp", "TestID", "Category", "Description", "InputPrompt",
            "ExpectedOutcomeCategory", "ExpectedScore",
            "ActualOutcomeCategory", "ActualScore",
            "ResultStatus", "DurationSec", "Notes"
        ]
        # Get all unique keys from data, ensuring preferred order comes first
        fieldnames_set = set()
        for row in all_results_data:
            fieldnames_set.update(row.keys())
        fieldnames = [f for f in preferred_order if f in fieldnames_set] + \
                     sorted([f for f in fieldnames_set if f not in preferred_order])

        try:
            with open(results_filename, 'w', newline='', encoding='utf-8') as csvfile:
                writer = csv.DictWriter(csvfile, fieldnames=fieldnames, extrasaction='ignore')
                writer.writeheader()
                writer.writerows(all_results_data)
            print(f"Results successfully written to {results_filename}")
        except Exception as e:
            print(f"Error writing results to CSV: {e}")
            logger.error(f"Failed to write results CSV: {e}")

    print("--- Financial Familiarity Testing Complete ---")
