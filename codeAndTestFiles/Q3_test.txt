        # --- Filtering Logic ---
        run_this_test = True
        if categories_to_run and test['category'] not in categories_to_run:
            run_this_test = False
        if ids_to_run and test_id not in ids_to_run:
            run_this_test = False

        if not run_this_test:
            results['skipped'] += 1
            continue
        # --- End Filtering ---

                     if bot_instance.validate_parsed_percentage(parsed_percentage, bot_instance.config):
                         percentage_result = parsed_percentage
                     else:
                         percentage_details = f"Overridden LLM output '{llm_raw_output}' parsed to {parsed_percentage}, but failed validation."
                         percentage_result = None # Mark as None due to validation fail
                else:
                    percentage_details = f"Overridden LLM output '{llm_raw_output}' failed to parse."
                    percentage_result = None
            else:
                # --- Normal Execution: Call process_user_input ---
                percentage_result = bot_instance.process_user_input(test['prompt'])
                # Note: We don't easily get the raw LLM output here unless we modify process_user_input

            duration = time.monotonic() - start_time
            # --- End LLM/Processing Logic ---


            # --- Process Result ---
            if percentage_result is not None:
                percentage_details = f"Processed Percentage: {percentage_result:.2f}%"
                # Calculate score using the bot's config
                score = calculate_percentage_score(percentage_result, bot_instance.config.percentage_score_brackets_simple)
                if score is not None:
                    actual_score = score
                    outcome_type = 'score'
                    percentage_details += f" -> Score: {score}"
                else:
                    percentage_details += " | Scoring Failed (returned None)"
                    outcome_type = 'fail' # Treat scoring failure as test failure
            elif 'llm_direct_output_override' not in test:
                 # Only log this if it wasn't an override test where failure was expected
                percentage_details = "process_user_input returned None (Likely LLM='None', parse error, or validation fail)"
                outcome_type = 'fail'
            # If it was an override test resulting in None, percentage_details is already set above.

        except Exception as e:
            duration = time.monotonic() - start_time if 'start_time' in locals() else -1.0
            logger.error(f"Exception during test case {test_id} ('{test['description']}'): {e}", exc_info=True)
            print(f"  ERROR: Exception occurred during processing: {e}")
            outcome_type = 'fail'
            percentage_details = f"Exception: {type(e).__name__}: {str(e)[:100]}..."
        # --- End Core Logic ---

        # --- Evaluate Result ---
        result_status = "FAIL"
        log_prefix = "🔴"

        if test['expected'] == outcome_type:
            result_status = "PASS"
            log_prefix = "✅"
            results['passed'] += 1
        elif test['expected'] == 'maybe_score' and outcome_type == 'score':
             result_status = f"PASS (Maybe - Matched Score)"
             log_prefix = "✅"
             results['passed'] += 1 # Count maybe_score that results in score as PASS
        elif test['expected'] == 'maybe_score' and outcome_type == 'fail':
            result_status = f"INFO (Maybe - Matched Fail)"
            log_prefix = "🤔"
            results['maybe'] += 1
        else: # Actual mismatch
            results['failed'] += 1

        print(f"  {log_prefix} Result: {result_status}")
        print(f"      Actual Outcome Type: {outcome_type}")
        print(f"      Duration: {duration:.4f} sec")
        print(f"      Processing Details: {percentage_details}")
        if actual_score is not None:
             print(f"      Actual Score: {actual_score}")

        # --- Store result for CSV ---
        result_row = {
            "RunTimestamp": datetime.datetime.now(datetime.timezone.utc).isoformat(),
            "TestID": test_id,
            "Category": test['category'],
            "Description": test['description'],
            "InputPrompt": test['prompt'],
            "ExpectedOutcome": test['expected'],
            "ActualOutcome": outcome_type,
            "ResultStatus": result_status.split(" ")[0], # Just PASS/FAIL/INFO
            "DurationSec": f"{duration:.4f}",
            "ActualPercentage": f"{percentage_result:.2f}" if percentage_result is not None else "",
            "ActualScore": str(actual_score) if actual_score is not None else "",
            "LLMRawOutputOverride": llm_raw_output if 'llm_direct_output_override' in test else "", # Log overridden output
            "Notes": percentage_details
        }
        all_results_data.append(result_row)

        # Optional: Add a small delay between API calls if needed, especially for free tiers
        # time.sleep(0.2)

    # --- Print Summary ---
    print("\n" + "="*30)
    print("--- Test Summary ---")
    print(f"Total Tests Defined: {total_tests}")
    print(f"Total Tests Skipped: {results['skipped']}")
    print(f"Total Tests Run: {results['total_run']}")
    print(f"✅ Passed: {results['passed']}")
    print(f"🔴 Failed: {results['failed']}")
    print(f"🤔 Maybe (Info - Fail): {results['maybe']}")
    print("="*30)

    # --- Write results to CSV ---
    print("\n--- Writing Test Results to CSV ---")
    results_filename = "income_percentage_test_run_latest.csv"
    if not all_results_data:
        print("No results data to write.")
    else:
        # Dynamically determine fieldnames but prefer a specific order
        fieldnames_set = set()
        for row in all_results_data:
            fieldnames_set.update(row.keys())
        preferred_order = ["RunTimestamp", "TestID", "Category", "Description", "InputPrompt",
                           "ExpectedOutcome", "ActualOutcome", "ResultStatus", "DurationSec",
                           "ActualPercentage", "ActualScore", "LLMRawOutputOverride", "Notes"]
        fieldnames = [f for f in preferred_order if f in fieldnames_set] + \
                     sorted([f for f in fieldnames_set if f not in preferred_order])

        try:
            with open(results_filename, 'w', newline='', encoding='utf-8') as csvfile:
                writer = csv.DictWriter(csvfile, fieldnames=fieldnames, extrasaction='ignore')
                writer.writeheader()
                writer.writerows(all_results_data)
            print(f"Results successfully written to {results_filename}")
        except Exception as e:
            print(f"Error writing results to CSV: {e}")
            logger.error(f"Failed to write results CSV: {e}")

    print("--- Testing Complete ---")
